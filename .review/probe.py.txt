print("== P1: merge real temporal objects into TOML ==")
doc = tomlkit.parse("[task]\nrun_at = 2026-11-16T23:39:55\nrun_on = 2026-11-16\nwindow = 23:39:55\n")
try:
    _merge_into_document(doc["task"], {
        "run_at": datetime(2026, 11, 16, 23, 39, 55),
        "run_on": date(2026, 11, 16),
        "window": time(23, 39, 55),
    })
    print("dumps ->", repr(tomlkit.dumps(doc)))
    print("reparsed types ->", {k: type(v).__name__ for k, v in doc["task"].items()})
except Exception as exc:
    print("EXC", type(exc).__name__, exc)

print("== P6: managed tree of BotConfig (unknown-key protection) ==")
from neobot_app.config.models import BotConfig  # noqa: E402
tree = _managed_tree(BotConfig)
from dataclasses import fields as dc_fields  # noqa: E402
from dataclasses import is_dataclass  # noqa: E402
from typing import get_args, get_origin  # noqa: E402
bad = []
for f in dc_fields(BotConfig):
    target = tree.get(f.name)
    ann = f.type
    if target is None and (is_dataclass(ann) or get_origin(ann) in (list, dict)):
        bad.append((f.name, str(ann), target))
print("tree entries:", len(tree), "suspicious:", bad)
