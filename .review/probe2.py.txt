"""PR325 probe 2: dataclass object-list path with cleared optional temporal."""
from __future__ import annotations

from dataclasses import dataclass, field
from datetime import datetime
from typing import Optional

import tomlkit

from neobot_app.builtin_plugins.dashboard.config_parts.schema import _coerce_payload, _validate_payload
from neobot_app.builtin_plugins.dashboard.config_parts.toml import (
    _diff_document,
    _managed_tree,
    _merge_into_document,
)


@dataclass
class Row:
    at: Optional[datetime] = None
    note: str = ""


@dataclass
class Root:
    rows: list[Row] = field(default_factory=list)


print("== A: new object-list row carrying a cleared optional temporal ==")
doc = tomlkit.parse("[[rows]]\nat = 2026-11-16T23:39:55\nnote = \"a\"\n")
current = doc.unwrap()
submitted = _coerce_payload(Root, {"rows": [{"at": "2026-11-16T23:39:55", "note": "a"}, {"at": "", "note": "b"}]})
print("coerced:", submitted)
errors: list[dict[str, str]] = []
_validate_payload(Root, submitted, "", errors)
print("errors:", errors)
diff = _diff_document(current, submitted, Root)
print("diff:", diff)
try:
    _merge_into_document(doc, diff)
    print("dumps ->", repr(tomlkit.dumps(doc)))
except Exception as exc:
    print("!! SAVE WOULD RAISE:", type(exc).__name__, exc)

print()
print("== B: same but clearing inside an EXISTING row ==")
doc2 = tomlkit.parse("[[rows]]\nat = 2026-11-16T23:39:55\nnote = \"a\"\n")
cur2 = doc2.unwrap()
sub2 = _coerce_payload(Root, {"rows": [{"at": "", "note": "a"}]})
diff2 = _diff_document(cur2, sub2, Root)
print("diff:", diff2)
_merge_into_document(doc2, diff2)
print("dumps ->", repr(tomlkit.dumps(doc2)))

print()
print("== C: managed tree for BotConfig ==")
from neobot_app.config.schemas.bot import BotConfig  # noqa: E402
tree = _managed_tree(BotConfig)
from dataclasses import fields as dc_fields, is_dataclass  # noqa: E402
from typing import get_origin  # noqa: E402

bad = []
for f in dc_fields(BotConfig):
    if tree.get(f.name) is None and (is_dataclass(f.type) or get_origin(f.type) in (list, dict)):
        bad.append((f.name, str(f.type)))
print("managed entries:", len(tree), "| degraded:", bad)
